1 · Specialists with guardrails and live tools

meridian/agents.py
from typing import Annotated
from pydantic import Field
from agent_framework import tool, MCPStreamableHTTPTool
from agent_framework.azure import AzureOpenAIChatClient

client = AzureOpenAIChatClient()

# Live systems via MCP (Lesson 09)
monitoring = MCPStreamableHTTPTool(name="monitoring", url="https://obs.meridian.internal/mcp")
runbooks = MCPStreamableHTTPTool(name="runbooks", url="https://kb.meridian.internal/mcp")

# Sensitive action gated for human approval (Lesson 17)
@tool(approval_mode="always_require")
def apply_remediation(
    change: Annotated[str, Field(description="The change to apply")],
) -> str:
    """Apply a production change. Requires human approval."""
    return f"Applied: {change}"

diagnostician = client.create_agent(
    name="Diagnostician",
    instructions="Find the root cause from metrics and recent deploys.",
    tools=[monitoring],
)
researcher = client.create_agent(
    name="Researcher",
    instructions="Match the incident to known runbooks and mitigations.",
    tools=[runbooks],
)
remediation = client.create_agent(
    name="Remediation",
    instructions="Propose and apply a minimal, reversible fix with rollback.",
    tools=[apply_remediation],
)

2 · Supervisor + memory, wrapped in middleware

meridian/workflow.py
from agent_framework import MagenticBuilder, ChatMessageStore
from .agents import client, diagnostician, researcher, remediation
from .memory import IncidentMemory          # ContextProvider (Lesson 18)
from .guards import AuthMiddleware, PiiRedactionMiddleware  # Lesson 10

def build_workflow():
    return (
        MagenticBuilder()
        .participants(
            diagnostician=diagnostician,
            researcher=researcher,
            remediation=remediation,
        )
        .with_standard_manager(
            chat_client=client,
            max_round_count=14,
            max_stall_count=3,
            max_reset_count=2,
        )
        .build()
    )

3 · Observability + the Azure Functions front door

function_app.py
from azure.monitor.opentelemetry import configure_azure_monitor
from agent_framework.observability import configure_otel_providers
from agent_framework.azure import AgentFunctionApp
from .workflow import build_workflow

configure_azure_monitor()       # Lesson 19 → Application Insights
configure_otel_providers()

# Expose the whole supervisor workflow as one durable agent endpoint (Lesson 22)
meridian = build_workflow().as_agent(name="Meridian")
app = AgentFunctionApp(agents=[meridian])
# POST /api/agents/Meridian/run

Make it yours

  1. Swap the domain — turn Meridian into loan processing, claims triage, or supply-chain ops. The skeleton holds.
  2. Add a sub-workflow — nest a reusable "compliance review" (Lesson 16) before remediation.
  3. Federate a partner — pull an external vendor agent in over A2A.
  4. Go declarative — move the agent definitions to YAML so ops can tune them without a deploy.

Production checklist
Before go-live: managed-identity auth, approval gates on every irreversible tool, OpenTelemetry to your APM, prompt/response logging policy reviewed, cost limits (max_round_count/max_reset_count), durable checkpoints, and load testing of your busiest path.
